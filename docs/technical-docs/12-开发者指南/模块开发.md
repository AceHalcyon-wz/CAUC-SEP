## 1. 概述

本文档面向CAUC-SEP自旋电子器件实验平台的开发者，详细介绍如何开发新模块并集成到现有系统中。模块开发遵循统一的架构模式和代码规范，确保系统的可维护性和可扩展性。

### 1.1 模块分类

| 模块类型 | 位置 | 说明 |
|----------|------|------|
| 设备驱动 | `backend/drivers/` | 硬件设备驱动程序 |
| API模块 | `backend/api/` | REST API路由模块 |
| 核心模块 | `backend/core/` | 核心业务逻辑 |
| 前端组件 | `frontend/src/components/` | Vue组件 |
| 组合式函数 | `frontend/src/composables/` | Vue Composition API函数 |

### 1.2 开发原则

1. **模块化设计**: 每个模块职责单一，高内聚低耦合
2. **接口抽象**: 通过接口定义模块边界，便于测试和替换
3. **配置驱动**: 通过配置文件控制模块行为，避免硬编码
4. **文档完整**: 每个模块必须包含完整的docstring文档

### 2.2 驱动基类

所有设备驱动必须继承 `BaseDriver` 类：

```python
"""
文件名: drivers/base.py
功能: 设备驱动基类定义
"""

from abc import ABC, abstractmethod
from typing import Optional, Dict, Any
from enum import Enum
from dataclasses import dataclass
import logging


class DeviceStatus(Enum):
    """设备状态枚举。"""
    DISCONNECTED = "disconnected"
    CONNECTED = "connected"
    READY = "ready"
    BUSY = "busy"
    ERROR = "error"


@dataclass
class DeviceInfo:
    """设备信息数据类。"""
    device_id: str
    device_type: str
    manufacturer: str
    model: str
    firmware_version: Optional[str] = None
    serial_number: Optional[str] = None


class BaseDriver(ABC):
    """
    设备驱动基类。

    所有设备驱动必须继承此类并实现所有抽象方法。
    """

    def __init__(self, device_id: str, config: Optional[Dict[str, Any]] = None):
        """
        初始化设备驱动。

        Args:
            device_id: 设备唯一标识符
            config: 设备配置字典
        """
        self.device_id = device_id
        self.config = config or {}
        self.status = DeviceStatus.DISCONNECTED
        self.logger = logging.getLogger(f"{self.__class__.__name__}.{device_id}")
        self._connection = None

    @abstractmethod
    async def connect(self) -> bool:
        """
        连接设备。

        Returns:
            连接是否成功
        """
        pass

    @abstractmethod
    async def disconnect(self) -> bool:
        """
        断开设备连接。

        Returns:
            断开是否成功
        """
        pass

    @abstractmethod
    async def get_info(self) -> DeviceInfo:
        """
        获取设备信息。

        Returns:
            设备信息对象
        """
        pass

    @abstractmethod
    async def configure(self, params: Dict[str, Any]) -> bool:
        """
        配置设备参数。

        Args:
            params: 配置参数字典

        Returns:
            配置是否成功
        """
        pass

    @abstractmethod
    async def read_status(self) -> Dict[str, Any]:
        """
        读取设备状态。

        Returns:
            状态信息字典
        """
        pass

    @abstractmethod
    async def execute_command(self, command: str, params: Optional[Dict[str, Any]] = None) -> Any:
        """
        执行设备命令。

        Args:
            command: 命令名称
            params: 命令参数

        Returns:
            命令执行结果
        """
        pass

    async def health_check(self) -> Dict[str, Any]:
        """
        设备健康检查。

        Returns:
            健康状态信息
        """
        try:
            status = await self.read_status()
            return {
                "device_id": self.device_id,
                "status": self.status.value,
                "healthy": self.status == DeviceStatus.READY,
                "last_check": status,
            }
        except Exception as e:
            self.logger.error(f"Health check failed: {e}")
            return {
                "device_id": self.device_id,
                "status": DeviceStatus.ERROR.value,
                "healthy": False,
                "error": str(e),
            }

    def __repr__(self) -> str:
        return f"{self.__class__.__name__}(device_id={self.device_id}, status={self.status.value})"
```

### 2.3 开发新设备驱动

以下以开发一个新的光谱仪驱动为例：

```python
"""
文件名: drivers/spectrometer.py
功能: 光谱仪设备驱动
"""

from typing import Optional, Dict, Any, List
import asyncio
import serial
import numpy as np

from drivers.base import BaseDriver, DeviceStatus, DeviceInfo


class SpectrometerDriver(BaseDriver):
    """
    光谱仪设备驱动。

    支持USB和串口连接的光谱仪设备。

    Example:
        >>> driver = SpectrometerDriver("spec_01", {"port": "COM3", "baud_rate": 115200})
        >>> await driver.connect()
        >>> spectrum = await driver.acquire_spectrum()
        >>> await driver.disconnect()
    """

    # 支持的波长范围
    MIN_WAVELENGTH = 200  # nm
    MAX_WAVELENGTH = 1100  # nm

    # 积分时间范围
    MIN_INTEGRATION_TIME = 1  # ms
    MAX_INTEGRATION_TIME = 10000  # ms

    def __init__(self, device_id: str, config: Optional[Dict[str, Any]] = None):
        """
        初始化光谱仪驱动。

        Args:
            device_id: 设备唯一标识符
            config: 配置参数，支持:
                - port: 串口号 (默认 "COM3")
                - baud_rate: 波特率 (默认 115200)
                - timeout: 超时时间秒 (默认 5.0)
        """
        super().__init__(device_id, config)

        self.port = config.get("port", "COM3")
        self.baud_rate = config.get("baud_rate", 115200)
        self.timeout = config.get("timeout", 5.0)

        # 光谱仪参数
        self._integration_time = 100  # ms
        self._wavelengths: Optional[np.ndarray] = None
        self._pixel_count = 2048

    async def connect(self) -> bool:
        """
        连接光谱仪设备。

        Returns:
            连接是否成功
        """
        try:
            # 创建串口连接
            self._connection = serial.Serial(
                port=self.port,
                baudrate=self.baud_rate,
                timeout=self.timeout,
            )

            # 等待设备响应
            await asyncio.sleep(0.5)

            # 发送初始化命令
            self._connection.write(b"*IDN?\n")
            response = self._connection.readline().decode().strip()

            if not response:
                raise ConnectionError("设备无响应")

            # 获取波长校准数据
            await self._load_wavelength_calibration()

            self.status = DeviceStatus.READY
            self.logger.info(f"光谱仪 {self.device_id} 连接成功")
            return True

        except Exception as e:
            self.status = DeviceStatus.ERROR
            self.logger.error(f"连接失败: {e}")
            return False

    async def disconnect(self) -> bool:
        """
        断开光谱仪连接。

        Returns:
            断开是否成功
        """
        try:
            if self._connection and self._connection.is_open:
                self._connection.close()

            self._connection = None
            self.status = DeviceStatus.DISCONNECTED
            self.logger.info(f"光谱仪 {self.device_id} 已断开")
            return True

        except Exception as e:
            self.logger.error(f"断开连接失败: {e}")
            return False

    async def get_info(self) -> DeviceInfo:
        """
        获取光谱仪设备信息。

        Returns:
            设备信息对象
        """
        if not self._connection:
            raise ConnectionError("设备未连接")

        self._connection.write(b"*IDN?\n")
        response = self._connection.readline().decode().strip()
        parts = response.split(",")

        return DeviceInfo(
            device_id=self.device_id,
            device_type="spectrometer",
            manufacturer=parts[0] if len(parts) > 0 else "Unknown",
            model=parts[1] if len(parts) > 1 else "Unknown",
            firmware_version=parts[2] if len(parts) > 2 else None,
            serial_number=parts[3] if len(parts) > 3 else None,
        )

    async def configure(self, params: Dict[str, Any]) -> bool:
        """
        配置光谱仪参数。

        Args:
            params: 配置参数，支持:
                - integration_time: 积分时间(ms)
                - scans_to_average: 平均扫描次数
                - boxcar_width: 平滑宽度

        Returns:
            配置是否成功
        """
        try:
            # 设置积分时间
            if "integration_time" in params:
                time_ms = params["integration_time"]
                if not self.MIN_INTEGRATION_TIME <= time_ms <= self.MAX_INTEGRATION_TIME:
                    raise ValueError(f"积分时间必须在 {self.MIN_INTEGRATION_TIME}-{self.MAX_INTEGRATION_TIME} ms 范围内")

                command = f"INT:{time_ms}\n"
                self._connection.write(command.encode())
                self._integration_time = time_ms

            # 设置平均扫描次数
            if "scans_to_average" in params:
                scans = params["scans_to_average"]
                command = f"SCANS:{scans}\n"
                self._connection.write(command.encode())

            # 设置平滑宽度
            if "boxcar_width" in params:
                width = params["boxcar_width"]
                command = f"BOXCAR:{width}\n"
                self._connection.write(command.encode())

            self.logger.info(f"光谱仪配置更新: {params}")
            return True

        except Exception as e:
            self.logger.error(f"配置失败: {e}")
            return False

    async def read_status(self) -> Dict[str, Any]:
        """
        读取光谱仪状态。

        Returns:
            状态信息字典
        """
        return {
            "device_id": self.device_id,
            "status": self.status.value,
            "integration_time": self._integration_time,
            "connected": self._connection is not None and self._connection.is_open,
            "port": self.port,
        }

    async def execute_command(self, command: str, params: Optional[Dict[str, Any]] = None) -> Any:
        """
        执行光谱仪命令。

        Args:
            command: 命令名称，支持:
                - acquire: 采集光谱
                - set_integration_time: 设置积分时间
                - get_wavelengths: 获取波长数组
            params: 命令参数

        Returns:
            命令执行结果
        """
        command_map = {
            "acquire": self.acquire_spectrum,
            "set_integration_time": self._set_integration_time,
            "get_wavelengths": self.get_wavelengths,
        }

        if command not in command_map:
            raise ValueError(f"未知命令: {command}")

        func = command_map[command]
        if params:
            return await func(**params)
        return await func()

    async def acquire_spectrum(self) -> Dict[str, Any]:
        """
        采集光谱数据。

        Returns:
            包含波长和强度数据的字典
        """
        if self.status != DeviceStatus.READY:
            raise RuntimeError("设备未就绪")

        self.status = DeviceStatus.BUSY

        try:
            # 发送采集命令
            self._connection.write(b"ACQUIRE\n")

            # 等待采集完成
            await asyncio.sleep(self._integration_time / 1000 + 0.1)

            # 读取数据
            data_bytes = self._connection.read(self._pixel_count * 2)
            intensities = np.frombuffer(data_bytes, dtype=np.uint16)

            # 转换为物理单位
            intensities_float = intensities.astype(np.float64)

            self.status = DeviceStatus.READY

            return {
                "wavelengths": self._wavelengths.tolist() if self._wavelengths is not None else None,
                "intensities": intensities_float.tolist(),
                "integration_time": self._integration_time,
                "pixel_count": self._pixel_count,
            }

        except Exception as e:
            self.status = DeviceStatus.ERROR
            self.logger.error(f"光谱采集失败: {e}")
            raise

    async def get_wavelengths(self) -> List[float]:
        """
        获取波长校准数组。

        Returns:
            波长数组(nm)
        """
        if self._wavelengths is None:
            await self._load_wavelength_calibration()
        return self._wavelengths.tolist()

    async def _set_integration_time(self, time_ms: int) -> bool:
        """
        设置积分时间。

        Args:
            time_ms: 积分时间(ms)

        Returns:
            设置是否成功
        """
        return await self.configure({"integration_time": time_ms})

    async def _load_wavelength_calibration(self) -> None:
        """
        加载波长校准数据。

        从设备读取或使用默认校准系数计算波长数组。
        """
        try:
            # 尝试从设备读取校准系数
            self._connection.write(b"WLCAL?\n")
            response = self._connection.readline().decode().strip()

            if response:
                # 解析校准系数: C0,C1,C2,C3
                coeffs = [float(x) for x in response.split(",")]

                # 计算波长数组: λ = C0 + C1*p + C2*p^2 + C3*p^3
                pixels = np.arange(self._pixel_count)
                self._wavelengths = (
                    coeffs[0]
                    + coeffs[1] * pixels
                    + coeffs[2] * pixels**2
                    + coeffs[3] * pixels**3
                )
            else:
                # 使用默认线性校准
                self._wavelengths = np.linspace(
                    self.MIN_WAVELENGTH, self.MAX_WAVELENGTH, self._pixel_count
                )

        except Exception as e:
            self.logger.warning(f"波长校准加载失败，使用默认值: {e}")
            self._wavelengths = np.linspace(
                self.MIN_WAVELENGTH, self.MAX_WAVELENGTH, self._pixel_count
            )
```

### 3.2 API模块开发规范

```python
"""
文件名: api/spectrometer.py
功能: 光谱仪API路由
"""

from typing import Optional, List
from fastapi import APIRouter, HTTPException, Depends, Query
from pydantic import BaseModel, Field
import logging

from drivers import create_driver, DRIVER_REGISTRY
from core.dependencies import get_device_manager

router = APIRouter(prefix="/api/v1/spectrometer", tags=["spectrometer"])
logger = logging.getLogger(__name__)


# ============== Pydantic 模型 ==============

class SpectrometerConfig(BaseModel):
    """光谱仪配置模型。"""
    integration_time: int = Field(
        default=100,
        description="积分时间(ms)",
        ge=1,
        le=10000,
    )
    scans_to_average: int = Field(
        default=1,
        description="平均扫描次数",
        ge=1,
        le=100,
    )
    boxcar_width: int = Field(
        default=0,
        description="平滑宽度",
        ge=0,
        le=10,
    )


class SpectrumData(BaseModel):
    """光谱数据模型。"""
    wavelengths: List[float] = Field(..., description="波长数组(nm)")
    intensities: List[float] = Field(..., description="强度数组")
    integration_time: int = Field(..., description="积分时间(ms)")
    pixel_count: int = Field(..., description="像素数")


# ============== API 端点 ==============

@router.get("/devices", summary="获取光谱仪设备列表")
async def list_spectrometers():
    """
    获取所有已注册的光谱仪设备。

    Returns:
        设备列表
    """
    device_manager = get_device_manager()
    spectrometers = [
        device for device in device_manager.list_devices()
        if device["type"] == "spectrometer"
    ]
    return {"devices": spectrometers, "count": len(spectrometers)}


@router.post("/{device_id}/connect", summary="连接光谱仪")
async def connect_spectrometer(device_id: str):
    """
    连接指定的光谱仪设备。

    Args:
        device_id: 设备ID

    Returns:
        连接结果
    """
    device_manager = get_device_manager()

    try:
        driver = device_manager.get_device(device_id)
        if not driver:
            raise HTTPException(status_code=404, detail=f"设备 {device_id} 不存在")

        success = await driver.connect()
        if not success:
            raise HTTPException(status_code=500, detail="连接失败")

        return {"device_id": device_id, "status": "connected", "success": True}

    except Exception as e:
        logger.error(f"连接光谱仪失败: {e}")
        raise HTTPException(status_code=500, detail=str(e))


@router.post("/{device_id}/disconnect", summary="断开光谱仪")
async def disconnect_spectrometer(device_id: str):
    """
    断开指定的光谱仪设备。

    Args:
        device_id: 设备ID

    Returns:
        断开结果
    """
    device_manager = get_device_manager()

    try:
        driver = device_manager.get_device(device_id)
        if not driver:
            raise HTTPException(status_code=404, detail=f"设备 {device_id} 不存在")

        success = await driver.disconnect()
        return {"device_id": device_id, "status": "disconnected", "success": success}

    except Exception as e:
        logger.error(f"断开光谱仪失败: {e}")
        raise HTTPException(status_code=500, detail=str(e))


@router.get("/{device_id}/info", summary="获取设备信息")
async def get_spectrometer_info(device_id: str):
    """
    获取光谱仪设备信息。

    Args:
        device_id: 设备ID

    Returns:
        设备信息
    """
    device_manager = get_device_manager()
    driver = device_manager.get_device(device_id)

    if not driver:
        raise HTTPException(status_code=404, detail=f"设备 {device_id} 不存在")

    info = await driver.get_info()
    return info.__dict__


@router.post("/{device_id}/configure", summary="配置光谱仪参数")
async def configure_spectrometer(device_id: str, config: SpectrometerConfig):
    """
    配置光谱仪参数。

    Args:
        device_id: 设备ID
        config: 配置参数

    Returns:
        配置结果
    """
    device_manager = get_device_manager()
    driver = device_manager.get_device(device_id)

    if not driver:
        raise HTTPException(status_code=404, detail=f"设备 {device_id} 不存在")

    success = await driver.configure(config.dict())
    return {"device_id": device_id, "success": success, "config": config.dict()}


@router.post("/{device_id}/acquire", summary="采集光谱数据")
async def acquire_spectrum(device_id: str) -> SpectrumData:
    """
    采集光谱数据。

    Args:
        device_id: 设备ID

    Returns:
        光谱数据
    """
    device_manager = get_device_manager()
    driver = device_manager.get_device(device_id)

    if not driver:
        raise HTTPException(status_code=404, detail=f"设备 {device_id} 不存在")

    try:
        result = await driver.acquire_spectrum()
        return SpectrumData(**result)

    except Exception as e:
        logger.error(f"光谱采集失败: {e}")
        raise HTTPException(status_code=500, detail=str(e))


@router.get("/{device_id}/wavelengths", summary="获取波长校准数据")
async def get_wavelengths(device_id: str):
    """
    获取波长校准数组。

    Args:
        device_id: 设备ID

    Returns:
        波长数组
    """
    device_manager = get_device_manager()
    driver = device_manager.get_device(device_id)

    if not driver:
        raise HTTPException(status_code=404, detail=f"设备 {device_id} 不存在")

    wavelengths = await driver.get_wavelengths()
    return {"device_id": device_id, "wavelengths": wavelengths}


@router.get("/{device_id}/status", summary="获取设备状态")
async def get_spectrometer_status(device_id: str):
    """
    获取光谱仪状态。

    Args:
        device_id: 设备ID

    Returns:
        状态信息
    """
    device_manager = get_device_manager()
    driver = device_manager.get_device(device_id)

    if not driver:
        raise HTTPException(status_code=404, detail=f"设备 {device_id} 不存在")

    status = await driver.read_status()
    return status
```
