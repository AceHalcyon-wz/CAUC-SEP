## 一、概述

CAUC-SEP 设备驱动层是硬件抽象层的核心，负责与物理设备通信。本文档定义了设备驱动的开发规范、接口标准和最佳实践。

### 1.1 驱动架构

```
┌─────────────────────────────────────────────────────────────────┐
│                      API层 (FastAPI路由)                         │
├─────────────────────────────────────────────────────────────────┤
│                      业务逻辑层 (Services)                        │
├─────────────────────────────────────────────────────────────────┤
│                      设备抽象层 (BaseDevice)                      │
├─────────────────────────────────────────────────────────────────┤
│                      驱动层 (Drivers)                             │
│  ┌───────────┐ ┌───────────┐ ┌───────────┐ ┌───────────┐      │
│  │ DM2C驱动   │ │ 电磁铁驱动 │ │ 温控器驱动  │ │ 压电驱动   │      │
│  └───────────┘ └───────────┘ └───────────┘ └───────────┘      │
├─────────────────────────────────────────────────────────────────┤
│                      通信层 (Modbus/Serial)                       │
└─────────────────────────────────────────────────────────────────┘
```

### 1.2 驱动职责

| 职责 | 说明 |
|-----|------|
| 设备连接 | 建立和维护与硬件设备的通信连接 |
| 状态读取 | 实时读取设备状态、参数 |
| 指令下发 | 发送控制指令到设备 |
| 异常处理 | 处理通信异常、设备报警 |
| 安全保护 | 实现软件限位、急停等安全机制 |

## 二、设备抽象基类

### 2.1 BaseDevice 接口定义

```python
"""
文件名: base.py
路径: backend/devices/base.py
功能: 设备抽象基类定义
"""

from abc import ABC, abstractmethod
from typing import Optional, Dict, Any
from enum import Enum
from dataclasses import dataclass
from datetime import datetime


class DeviceStatus(str, Enum):
    """设备状态枚举。"""
    DISCONNECTED = "disconnected"
    CONNECTING = "connecting"
    CONNECTED = "connected"
    ERROR = "error"
    ALARM = "alarm"


@dataclass
class DeviceInfo:
    """设备信息数据类。"""
    device_id: str
    device_type: str
    manufacturer: str
    model: str
    firmware_version: Optional[str] = None
    serial_number: Optional[str] = None


@dataclass
class DeviceAlarm:
    """设备报警数据类。"""
    alarm_code: int
    alarm_message: str
    alarm_level: str  # "warning", "error", "critical"
    timestamp: datetime


class BaseDevice(ABC):
    """
    设备抽象基类。
    
    所有设备驱动必须继承此类并实现所有抽象方法。
    提供统一的设备管理接口，支持设备连接、状态读取、
    指令下发、异常处理等功能。
    
    安全约束:
        - 所有驱动必须实现急停功能
        - 所有驱动必须处理通信异常
        - 所有驱动必须记录操作日志
    """
    
    def __init__(self, device_id: str, config: Dict[str, Any]):
        """
        初始化设备实例。
        
        Args:
            device_id: 设备唯一标识
            config: 设备配置字典
        """
        self.device_id = device_id
        self.config = config
        self._status = DeviceStatus.DISCONNECTED
        self._alarm: Optional[DeviceAlarm] = None
        self._last_update: Optional[datetime] = None
    
    @property
    def status(self) -> DeviceStatus:
        """获取设备状态。"""
        return self._status
    
    @property
    def is_connected(self) -> bool:
        """检查设备是否已连接。"""
        return self._status == DeviceStatus.CONNECTED
    
    @property
    def is_alarm(self) -> bool:
        """检查设备是否处于报警状态。"""
        return self._status == DeviceStatus.ALARM
    
    # ==================== 抽象方法（必须实现） ====================
    
    @abstractmethod
    async def connect(self) -> bool:
        """
        连接设备。
        
        Returns:
            连接是否成功
        
        Raises:
            DeviceConnectionError: 连接失败时抛出
        """
        pass
    
    @abstractmethod
    async def disconnect(self) -> bool:
        """
        断开设备连接。
        
        Returns:
            断开是否成功
        """
        pass
    
    @abstractmethod
    async def get_status(self) -> Dict[str, Any]:
        """
        获取设备状态信息。
        
        Returns:
            包含设备状态的字典
        
        Raises:
            DeviceCommunicationError: 通信失败时抛出
        """
        pass
    
    @abstractmethod
    async def emergency_stop(self) -> bool:
        """
        执行紧急停止。
        
        安全约束:
            - 此方法必须具有最高优先级
            - 必须立即停止所有运动
            - 必须记录急停操作日志
        
        Returns:
            急停是否成功
        
        Raises:
            DeviceEmergencyStopError: 急停失败时抛出
        """
        pass
    
    @abstractmethod
    async def reset_alarm(self) -> bool:
        """
        复位报警状态。
        
        Returns:
            复位是否成功
        """
        pass
    
    @abstractmethod
    async def get_device_info(self) -> DeviceInfo:
        """
        获取设备信息。
        
        Returns:
            设备信息对象
        """
        pass
    
    # ==================== 可选方法（按需实现） ====================
    
    async def health_check(self) -> bool:
        """
        设备健康检查。
        
        Returns:
            设备是否健康
        """
        try:
            status = await self.get_status()
            return status is not None
        except Exception:
            return False
    
    async def self_test(self) -> Dict[str, Any]:
        """
        设备自检。
        
        Returns:
            自检结果字典
        """
        return {
            "device_id": self.device_id,
            "status": self._status.value,
            "connected": self.is_connected,
            "alarm": self._alarm.alarm_message if self._alarm else None,
            "timestamp": datetime.now().isoformat()
        }
```

### 2.2 实现示例：步进电机驱动

```python
"""
文件名: stepper.py
路径: backend/devices/stepper.py
功能: 步进电机设备驱动实现
"""

import asyncio
from typing import Dict, Any, Optional
from datetime import datetime

from .base import BaseDevice, DeviceStatus, DeviceInfo, DeviceAlarm
from core.dm2c_driver import DM2CDriver
from core.exceptions import DeviceConnectionError, DeviceCommunicationError


class StepperMotorDevice(BaseDevice):
    """
    步进电机设备驱动。
    
    支持雷赛DM2C系列步进电机驱动器，提供运动控制、
    PR路径编程、IO端口配置等功能。
    
    安全约束:
        - 运动指令必须先进行软件限位校验
        - 急停指令具有最高优先级
        - 通信异常时自动锁止设备
    """
    
    # 设备类型标识
    DEVICE_TYPE = "stepper_motor"
    MANUFACTURER = "Leadshine"
    MODEL = "DM2C"
    
    def __init__(self, device_id: str, config: Dict[str, Any]):
        """
        初始化步进电机驱动。
        
        Args:
            device_id: 设备唯一标识
            config: 设备配置，包含串口参数、限位参数等
        """
        super().__init__(device_id, config)
        
        # 驱动实例
        self._driver: Optional[DM2CDriver] = None
        
        # 软件限位参数
        self._positive_limit = config.get("positive_limit", 50000)
        self._negative_limit = config.get("negative_limit", -50000)
        
        # 仿真模式
        self._simulation = config.get("simulation", False)
    
    async def connect(self) -> bool:
        """
        连接步进电机驱动器。
        
        Returns:
            连接是否成功
        
        Raises:
            DeviceConnectionError: 连接失败时抛出
        """
        try:
            self._status = DeviceStatus.CONNECTING
            
            # 创建驱动实例
            self._driver = DM2CDriver(
                port=self.config.get("port", "COM3"),
                slave_id=self.config.get("slave_id", 1),
                baudrate=self.config.get("baudrate", 115200),
                simulation=self._simulation
            )
            
            # 建立连接
            success = await self._driver.connect()
            
            if success:
                self._status = DeviceStatus.CONNECTED
                self._last_update = datetime.now()
                return True
            else:
                self._status = DeviceStatus.ERROR
                raise DeviceConnectionError(f"设备连接失败: {self.device_id}")
                
        except Exception as e:
            self._status = DeviceStatus.ERROR
            raise DeviceConnectionError(f"连接异常: {str(e)}")
    
    async def disconnect(self) -> bool:
        """断开设备连接。"""
        if self._driver:
            await self._driver.disconnect()
            self._driver = None
        self._status = DeviceStatus.DISCONNECTED
        return True
    
    async def get_status(self) -> Dict[str, Any]:
        """
        获取电机状态。
        
        Returns:
            包含位置、速度、状态字、报警代码的字典
        """
        if not self.is_connected or not self._driver:
            raise DeviceCommunicationError("设备未连接")
        
        try:
            status = await self._driver.read_all_status()
            self._last_update = datetime.now()
            
            # 检查报警状态
            if status.get("alarm_code", 0) != 0:
                self._status = DeviceStatus.ALARM
                self._alarm = DeviceAlarm(
                    alarm_code=status["alarm_code"],
                    alarm_message=self._get_alarm_message(status["alarm_code"]),
                    alarm_level="error",
                    timestamp=datetime.now()
                )
            
            return status
            
        except Exception as e:
            self._status = DeviceStatus.ERROR
            raise DeviceCommunicationError(f"状态读取失败: {str(e)}")
    
    async def emergency_stop(self) -> bool:
        """
        执行紧急停止。
        
        安全约束:
            - 立即停止所有运动
            - 记录急停操作日志
            - 锁止设备状态
        
        Returns:
            急停是否成功
        """
        if not self._driver:
            return False
        
        try:
            # 急停指令具有最高优先级
            success = await self._driver.emergency_stop()
            
            if success:
                self._status = DeviceStatus.ALARM
                self._alarm = DeviceAlarm(
                    alarm_code=0xFF,
                    alarm_message="用户触发急停",
                    alarm_level="critical",
                    timestamp=datetime.now()
                )
            
            return success
            
        except Exception as e:
            raise DeviceCommunicationError(f"急停执行失败: {str(e)}")
    
    async def reset_alarm(self) -> bool:
        """复位报警状态。"""
        if not self._driver:
            return False
        
        success = await self._driver.reset_alarm()
        if success:
            self._status = DeviceStatus.CONNECTED
            self._alarm = None
        
        return success
    
    async def get_device_info(self) -> DeviceInfo:
        """获取设备信息。"""
        return DeviceInfo(
            device_id=self.device_id,
            device_type=self.DEVICE_TYPE,
            manufacturer=self.MANUFACTURER,
            model=self.MODEL
        )
    
    # ==================== 步进电机专用方法 ====================
    
    async def absolute_move(
        self, 
        target_position: int, 
        speed: int,
        acceleration: int = 100
    ) -> bool:
        """
        执行绝对定位运动。
        
        Args:
            target_position: 目标位置（脉冲）
            speed: 运动速度（脉冲/秒）
            acceleration: 加速时间（毫秒）
        
        Returns:
            运动指令是否下发成功
        
        Raises:
            ValueError: 目标位置超出软件限位
            DeviceAlarmError: 设备处于报警状态
        """
        # 安全校验：软件限位检查
        if not self._check_position_in_limits(target_position):
            raise ValueError(
                f"目标位置 {target_position} 超出软件限位范围 "
                f"[{self._negative_limit}, {self._positive_limit}]"
            )
        
        # 状态检查：报警状态下禁止运动
        if self.is_alarm:
            raise DeviceAlarmError("设备处于报警状态，请先复位报警")
        
        if not self._driver:
            raise DeviceCommunicationError("设备未连接")
        
        return await self._driver.absolute_move(
            target_position, speed, acceleration
        )
    
    def _check_position_in_limits(self, position: int) -> bool:
        """检查位置是否在软件限位范围内。"""
        return self._negative_limit <= position <= self._positive_limit
    
    def _get_alarm_message(self, alarm_code: int) -> str:
        """获取报警代码对应的报警信息。"""
        alarm_messages = {
            0x01: "过流保护",
            0x02: "过压保护",
            0x03: "欠压保护",
            0x04: "过热保护",
            0x05: "编码器错误",
            0x06: "通信错误",
            # ... 更多报警代码
        }
        return alarm_messages.get(alarm_code, f"未知报警 (0x{alarm_code:02X})")
```

```python
from core.exceptions import (
    DeviceConnectionError,
    DeviceCommunicationError,
    DeviceAlarmError,
    DeviceEmergencyStopError,
    DeviceParameterError,
)

async def some_operation(self):
    """示例：异常处理规范。"""
    try:
        # 1. 前置校验
        if not self.is_connected:
            raise DeviceCommunicationError("设备未连接")
        
        # 2. 参数校验
        if not self._validate_parameters(params):
            raise DeviceParameterError("参数校验失败")
        
        # 3. 执行操作
        result = await self._driver.execute(params)
        
        # 4. 记录日志
        self._logger.info(f"操作成功: {result}")
        
        return result
        
    except DeviceCommunicationError as e:
        # 通信异常：记录日志，更新状态
        self._logger.error(f"通信异常: {e}")
        self._status = DeviceStatus.ERROR
        raise
        
    except DeviceAlarmError as e:
        # 报警异常：更新报警状态
        self._logger.warning(f"设备报警: {e}")
        self._status = DeviceStatus.ALARM
        raise
        
    except Exception as e:
        # 未知异常：记录完整堆栈
        self._logger.exception(f"未知异常: {e}")
        self._status = DeviceStatus.ERROR
        raise DeviceCommunicationError(f"操作失败: {str(e)}")
```

### 4.2 通信超时与重试

```python
class ModbusCommunication:
    """Modbus通信管理。"""
    
    # 通信参数
    DEFAULT_TIMEOUT = 1.0           # 默认超时（秒）
    MAX_RETRIES = 3                 # 最大重试次数
    RETRY_DELAY = 0.1               # 重试延迟（秒）
    
    async def read_registers_with_retry(
        self,
        address: int,
        count: int,
        timeout: float = DEFAULT_TIMEOUT
    ) -> list[int]:
        """
        带重试机制的寄存器读取。
        
        Args:
            address: 起始地址
            count: 寄存器数量
            timeout: 超时时间
        
        Returns:
            寄存器值列表
        
        Raises:
            ModbusCommunicationError: 通信失败时抛出
        """
        last_error = None
        
        for attempt in range(self.MAX_RETRIES):
            try:
                result = await self._client.read_holding_registers(
                    address=address,
                    count=count,
                    slave=self._slave_id
                )
                
                if not result.isError():
                    return result.registers
                
                last_error = f"Modbus错误: {result}"
                
            except Exception as e:
                last_error = str(e)
            
            # 重试延迟（指数退避）
            if attempt < self.MAX_RETRIES - 1:
                await asyncio.sleep(self.RETRY_DELAY * (2 ** attempt))
        
        raise ModbusCommunicationError(
            f"读取寄存器失败 (地址: 0x{address:04X}): {last_error}"
        )
```

## 六、驱动注册

### 6.1 设备注册表

```python
"""
文件名: device_registry.py
路径: backend/core/device_management/device_registry.py
功能: 设备注册表，管理所有设备实例
"""

from typing import Dict, Type, Optional
from devices.base import BaseDevice


class DeviceRegistry:
    """
    设备注册表。
    
    管理所有设备实例的创建、存储和访问。
    """
    
    _instance: Optional['DeviceRegistry'] = None
    _devices: Dict[str, BaseDevice] = {}
    _device_classes: Dict[str, Type[BaseDevice]] = {}
    
    def __new__(cls):
        """单例模式。"""
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance
    
    @classmethod
    def register_device_class(cls, device_type: str, device_class: Type[BaseDevice]):
        """
        注册设备类。
        
        Args:
            device_type: 设备类型标识
            device_class: 设备类
        """
        cls._device_classes[device_type] = device_class
    
    @classmethod
    def create_device(
        cls, 
        device_type: str, 
        device_id: str, 
        config: Dict
    ) -> BaseDevice:
        """
        创建设备实例。
        
        Args:
            device_type: 设备类型
            device_id: 设备ID
            config: 设备配置
        
        Returns:
            设备实例
        
        Raises:
            ValueError: 未知的设备类型
        """
        if device_type not in cls._device_classes:
            raise ValueError(f"未知的设备类型: {device_type}")
        
        device_class = cls._device_classes[device_type]
        device = device_class(device_id, config)
        cls._devices[device_id] = device
        
        return device
    
    @classmethod
    def get_device(cls, device_id: str) -> Optional[BaseDevice]:
        """获取设备实例。"""
        return cls._devices.get(device_id)
    
    @classmethod
    def get_all_devices(cls) -> Dict[str, BaseDevice]:
        """获取所有设备实例。"""
        return cls._devices.copy()


# 设备类注册装饰器
def register_device(device_type: str):
    """
    设备类注册装饰器。
    
    使用方法:
        @register_device("stepper_motor")
        class StepperMotorDevice(BaseDevice):
            ...
    """
    def decorator(cls):
        DeviceRegistry.register_device_class(device_type, cls)
        return cls
    return decorator
```
