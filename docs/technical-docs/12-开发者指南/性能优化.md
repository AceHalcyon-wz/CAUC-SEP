```python
"""
文件名: core/async_optimization.py
功能: 异步编程优化示例
"""

import asyncio
import aiohttp
from typing import List, Dict, Any
import logging

logger = logging.getLogger(__name__)


class AsyncDeviceController:
    """
    异步设备控制器。

    使用async/await实现非阻塞设备操作。
    """

    def __init__(self):
        """初始化控制器。"""
        self._session: aiohttp.ClientSession = None

    async def initialize(self) -> None:
        """初始化HTTP会话。"""
        self._session = aiohttp.ClientSession(
            timeout=aiohttp.ClientTimeout(total=30),
            connector=aiohttp.TCPConnector(limit=10),
        )

    async def close(self) -> None:
        """关闭HTTP会话。"""
        if self._session:
            await self._session.close()

    async def read_device(self, device_id: str) -> Dict[str, Any]:
        """
        异步读取设备数据。

        Args:
            device_id: 设备ID

        Returns:
            设备数据
        """
        url = f"http://device-server/{device_id}/read"

        async with self._session.get(url) as response:
            return await response.json()

    async def read_multiple_devices(
        self, device_ids: List[str]
    ) -> List[Dict[str, Any]]:
        """
        并发读取多个设备数据。

        Args:
            device_ids: 设备ID列表

        Returns:
            设备数据列表
        """
        # 使用asyncio.gather并发执行
        tasks = [self.read_device(device_id) for device_id in device_ids]
        results = await asyncio.gather(*tasks, return_exceptions=True)

        # 处理异常
        data = []
        for i, result in enumerate(results):
            if isinstance(result, Exception):
                logger.error(f"读取设备 {device_ids[i]} 失败: {result}")
                data.append({"device_id": device_ids[i], "error": str(result)})
            else:
                data.append(result)

        return data


# 使用示例
async def main():
    """主函数示例。"""
    controller = AsyncDeviceController()
    await controller.initialize()

    try:
        # 并发读取多个设备
        device_ids = ["dev_01", "dev_02", "dev_03", "dev_04"]
        results = await controller.read_multiple_devices(device_ids)
        print(f"读取完成: {len(results)} 个设备")
    finally:
        await controller.close()


# 运行
if __name__ == "__main__":
    asyncio.run(main())
```

```python
"""
文件名: core/concurrency.py
功能: 并发控制工具
"""

import asyncio
from typing import Callable, TypeVar, List, Any
from dataclasses import dataclass

T = TypeVar('T')


@dataclass
class ConcurrencyConfig:
    """并发配置。"""
    max_concurrent: int = 10  # 最大并发数
    batch_size: int = 100     # 批处理大小
    timeout: float = 30.0     # 超时时间(秒)


class ConcurrencyLimiter:
    """
    并发限制器。

    控制并发任务数量，防止资源耗尽。
    """

    def __init__(self, max_concurrent: int = 10):
        """
        初始化限制器。

        Args:
            max_concurrent: 最大并发数
        """
        self._semaphore = asyncio.Semaphore(max_concurrent)

    async def run_with_limit(
        self,
        func: Callable[..., T],
        *args,
        **kwargs,
    ) -> T:
        """
        在限制下运行函数。

        Args:
            func: 异步函数
            args: 位置参数
            kwargs: 关键字参数

        Returns:
            函数返回值
        """
        async with self._semaphore:
            return await func(*args, **kwargs)

    async def run_batch(
        self,
        func: Callable[..., T],
        items: List[Any],
        **kwargs,
    ) -> List[T]:
        """
        批量运行函数。

        Args:
            func: 异步函数
            items: 参数列表
            kwargs: 额外参数

        Returns:
            结果列表
        """
        tasks = [
            self.run_with_limit(func, item, **kwargs)
            for item in items
        ]
        return await asyncio.gather(*tasks, return_exceptions=True)


# 使用示例
async def process_item(item: dict) -> dict:
    """处理单个项目。"""
    await asyncio.sleep(0.1)  # 模拟处理
    return {"processed": item["id"]}


async def batch_process_example():
    """批处理示例。"""
    limiter = ConcurrencyLimiter(max_concurrent=5)

    items = [{"id": i} for i in range(100)]
    results = await limiter.run_batch(process_item, items)

    print(f"处理完成: {len(results)} 个项目")
```

#### 2.2.1 流式数据处理

```python
"""
文件名: core/stream_processor.py
功能: 流式数据处理
"""

import asyncio
from typing import AsyncIterator, Callable, Optional
import numpy as np
from dataclasses import dataclass
import logging

logger = logging.getLogger(__name__)


@dataclass
class StreamConfig:
    """流处理配置。"""
    buffer_size: int = 10000      # 缓冲区大小
    batch_size: int = 1000        # 批处理大小
    flush_interval: float = 1.0   # 刷新间隔(秒)


class StreamProcessor:
    """
    流式数据处理器。

    支持大数据集的流式处理，避免内存溢出。
    """

    def __init__(self, config: Optional[StreamConfig] = None):
        """
        初始化处理器。

        Args:
            config: 配置对象
        """
        self.config = config or StreamConfig()
        self._buffer: list = []
        self._total_processed = 0

    async def process_stream(
        self,
        data_stream: AsyncIterator[np.ndarray],
        process_func: Callable[[np.ndarray], np.ndarray],
    ) -> AsyncIterator[np.ndarray]:
        """
        处理数据流。

        Args:
            data_stream: 数据流
            process_func: 处理函数

        Yields:
            处理后的数据
        """
        async for chunk in data_stream:
            # 处理数据块
            processed = process_func(chunk)

            # 添加到缓冲区
            self._buffer.extend(processed)

            # 缓冲区满时输出
            if len(self._buffer) >= self.config.batch_size:
                yield np.array(self._buffer)
                self._buffer.clear()

            self._total_processed += len(chunk)

        # 输出剩余数据
        if self._buffer:
            yield np.array(self._buffer)
            self._buffer.clear()

    @staticmethod
    async def create_data_stream(
        data: np.ndarray,
        chunk_size: int = 1000,
    ) -> AsyncIterator[np.ndarray]:
        """
        创建数据流。

        Args:
            data: 数据数组
            chunk_size: 块大小

        Yields:
            数据块
        """
        for i in range(0, len(data), chunk_size):
            yield data[i:i + chunk_size]
            await asyncio.sleep(0)  # 让出控制权


# 使用示例
async def stream_processing_example():
    """流式处理示例。"""
    # 创建大数据集
    large_data = np.random.randn(100000)

    # 创建处理器
    processor = StreamProcessor(StreamConfig(batch_size=5000))

    # 定义处理函数
    def smooth(data: np.ndarray) -> np.ndarray:
        """平滑处理。"""
        window = 5
        return np.convolve(data, np.ones(window) / window, mode='valid')

    # 创建数据流
    data_stream = processor.create_data_stream(large_data, chunk_size=1000)

    # 处理数据流
    async for result in processor.process_stream(data_stream, smooth):
        print(f"处理数据块: {len(result)} 个点")
```

#### 2.2.2 内存优化

```python
"""
文件名: core/memory_optimization.py
功能: 内存优化工具
"""

import gc
import sys
from typing import Generator, Any
import numpy as np
from contextlib import contextmanager
import logging

logger = logging.getLogger(__name__)


class MemoryOptimizer:
    """
    内存优化工具类。

    提供内存监控和优化功能。
    """

    @staticmethod
    def get_memory_usage() -> dict:
        """
        获取当前内存使用情况。

        Returns:
            内存使用信息
        """
        import psutil
        import os

        process = psutil.Process(os.getpid())
        mem_info = process.memory_info()

        return {
            "rss_mb": mem_info.rss / 1024 / 1024,  # 物理内存
            "vms_mb": mem_info.vms / 1024 / 1024,  # 虚拟内存
            "percent": process.memory_percent(),    # 百分比
        }

    @staticmethod
    def optimize_numpy_array(arr: np.ndarray) -> np.ndarray:
        """
        优化NumPy数组内存使用。

        Args:
            arr: 输入数组

        Returns:
            优化后的数组
        """
        # 检查是否可以降低数据类型精度
        if arr.dtype == np.float64:
            # 检查是否可以用float32
            if np.allclose(arr, arr.astype(np.float32)):
                return arr.astype(np.float32)

        if arr.dtype == np.int64:
            # 检查是否可以用更小的整数类型
            min_val, max_val = arr.min(), arr.max()

            if min_val >= 0:
                if max_val < 256:
                    return arr.astype(np.uint8)
                elif max_val < 65536:
                    return arr.astype(np.uint16)
                elif max_val < 4294967296:
                    return arr.astype(np.uint32)
            else:
                if min_val >= -128 and max_val < 128:
                    return arr.astype(np.int8)
                elif min_val >= -32768 and max_val < 32768:
                    return arr.astype(np.int16)
                elif min_val >= -2147483648 and max_val < 2147483648:
                    return arr.astype(np.int32)

        return arr

    @staticmethod
    @contextmanager
    def memory_efficient_context():
        """
        内存高效上下文管理器。

        自动清理临时对象。
        """
        try:
            yield
        finally:
            gc.collect()

    @staticmethod
    def chunk_generator(
        data: np.ndarray,
        chunk_size: int,
    ) -> Generator[np.ndarray, None, None]:
        """
        分块生成器。

        避免一次性加载大数据集。

        Args:
            data: 数据数组
            chunk_size: 块大小

        Yields:
            数据块
        """
        for i in range(0, len(data), chunk_size):
            yield data[i:i + chunk_size]


# 使用示例
def memory_optimization_example():
    """内存优化示例。"""
    # 创建大数据集
    large_data = np.random.randn(1000000)

    # 优化内存
    optimized = MemoryOptimizer.optimize_numpy_array(large_data)

    print(f"原始大小: {sys.getsizeof(large_data) / 1024 / 1024:.2f} MB")
    print(f"优化大小: {sys.getsizeof(optimized) / 1024 / 1024:.2f} MB")

    # 使用内存高效上下文
    with MemoryOptimizer.memory_efficient_context():
        # 处理数据
        result = np.mean(optimized)
        print(f"计算结果: {result}")

    # 查看内存使用
    mem_usage = MemoryOptimizer.get_memory_usage()
    print(f"内存使用: {mem_usage['rss_mb']:.2f} MB")
```

### 2.3 启动优化

```python
"""
文件名: core/startup_optimization.py
功能: 启动优化
"""

import sys
import gc
import importlib
from typing import List, Optional
import logging

logger = logging.getLogger(__name__)


class StartupOptimizer:
    """
    启动优化器。

    优化应用启动性能。
    """

    def __init__(self):
        """初始化优化器。"""
        self._preloaded_modules: List[str] = []

    def preload_modules(self, modules: List[str]) -> None:
        """
        预加载模块。

        Args:
            modules: 模块名称列表
        """
        for module_name in modules:
            try:
                importlib.import_module(module_name)
                self._preloaded_modules.append(module_name)
                logger.debug(f"预加载模块: {module_name}")
            except ImportError as e:
                logger.warning(f"预加载模块失败 {module_name}: {e}")

    def optimize_gc(self) -> None:
        """
        优化垃圾回收器。

        调整GC参数以提高性能。
        """
        # 禁用自动GC（在启动期间）
        gc.disable()

        # 执行一次完整回收
        gc.collect()

        # 调整GC阈值
        # (threshold0, threshold1, threshold2)
        # threshold0: 触发0代回收的阈值
        # threshold1: 触发1代回收的阈值
        # threshold2: 触发2代回收的阈值
        gc.set_threshold(700, 10, 5)

        # 重新启用GC
        gc.enable()

        logger.info("GC优化完成")

    def optimize_numpy(self) -> None:
        """
        优化NumPy配置。

        设置NumPy使用优化的BLAS库。
        """
        try:
            import numpy as np

            # 设置打印选项（减少输出开销）
            np.set_printoptions(
                precision=4,
                threshold=1000,
                edgeitems=10,
            )

            # 检查BLAS配置
            blas_info = np.show_config(mode='dicts').get('blas_opt', {})
            logger.info(f"NumPy BLAS配置: {blas_info.get('name', 'unknown')}")

        except Exception as e:
            logger.warning(f"NumPy优化失败: {e}")

    def optimize_startup(
        self,
        preload: Optional[List[str]] = None,
        optimize_gc: bool = True,
        optimize_numpy: bool = True,
    ) -> dict:
        """
        执行启动优化。

        Args:
            preload: 预加载模块列表
            optimize_gc: 是否优化GC
            optimize_numpy: 是否优化NumPy

        Returns:
            优化结果
        """
        result = {
            "preloaded_modules": [],
            "gc_optimized": False,
            "numpy_optimized": False,
        }

        # 预加载模块
        if preload:
            self.preload_modules(preload)
            result["preloaded_modules"] = self._preloaded_modules

        # 优化GC
        if optimize_gc:
            self.optimize_gc()
            result["gc_optimized"] = True

        # 优化NumPy
        if optimize_numpy:
            self.optimize_numpy()
            result["numpy_optimized"] = True

        return result


# 全局优化函数
def optimize_startup(
    preload: Optional[List[str]] = None,
    optimize_gc: bool = True,
    optimize_numpy: bool = True,
) -> dict:
    """
    全局启动优化函数。

    Args:
        preload: 预加载模块列表
        optimize_gc: 是否优化GC
        optimize_numpy: 是否优化NumPy

    Returns:
        优化结果
    """
    optimizer = StartupOptimizer()
    return optimizer.optimize_startup(preload, optimize_gc, optimize_numpy)
```

```vue
<script setup lang="ts">
/**
 * @file VirtualList.vue
 * @description 虚拟滚动列表，优化大数据列表渲染性能
 */

import { ref, computed, onMounted, onUnmounted, watch } from 'vue';

interface Props {
  items: any[];
  itemHeight: number;
  visibleCount?: number;
  bufferSize?: number;
}

const props = withDefaults(defineProps<Props>(), {
  visibleCount: 20,
  bufferSize: 5,
});

const emit = defineEmits<{
  (e: 'item-click', item: any, index: number): void;
  (e: 'scroll', scrollTop: number): void;
}>();

// 容器引用
const containerRef = ref<HTMLElement>();

// 滚动状态
const scrollTop = ref(0);
const containerHeight = ref(0);

// 计算属性
const totalHeight = computed(() => props.items.length * props.itemHeight);

const startIndex = computed(() => {
  const index = Math.floor(scrollTop.value / props.itemHeight) - props.bufferSize;
  return Math.max(0, index);
});

const endIndex = computed(() => {
  const visibleCount = Math.ceil(containerHeight.value / props.itemHeight);
  const index = startIndex.value + visibleCount + props.bufferSize * 2;
  return Math.min(props.items.length, index);
});

const visibleItems = computed(() => {
  return props.items.slice(startIndex.value, endIndex.value).map((item, i) => ({
    item,
    index: startIndex.value + i,
  }));
});

const offsetY = computed(() => startIndex.value * props.itemHeight);

// 方法
function handleScroll(event: Event) {
  const target = event.target as HTMLElement;
  scrollTop.value = target.scrollTop;
  emit('scroll', scrollTop.value);
}

function scrollToIndex(index: number) {
  if (containerRef.value) {
    containerRef.value.scrollTop = index * props.itemHeight;
  }
}

// 生命周期
onMounted(() => {
  if (containerRef.value) {
    containerHeight.value = containerRef.value.clientHeight;
  }
});

// 暴露方法
defineExpose({
  scrollToIndex,
});
</script>

<template>
  <div
    ref="containerRef"
    class="virtual-list"
    @scroll="handleScroll"
  >
    <div
      class="virtual-list-content"
      :style="{ height: `${totalHeight}px` }"
    >
      <div
        class="virtual-list-items"
        :style="{ transform: `translateY(${offsetY}px)` }"
      >
        <div
          v-for="{ item, index } in visibleItems"
```

```typescript
/**
 * 智能数据采样
 * 根据数据量自动选择采样策略
 */
export function smartSampling(
  data: number[],
  maxPoints: number = 5000
): number[] {
  if (data.length <= maxPoints) {
    return data;
  }

  const ratio = data.length / maxPoints;
  const sampled: number[] = [];

  for (let i = 0; i < maxPoints; i++) {
    const index = Math.floor(i * ratio);
    sampled.push(data[index]);
  }

  return sampled;
}

/**
 * LTTB采样算法
 * 保留数据趋势特征的降采样
 */
export function lttbSampling(
  xData: number[],
  yData: number[],
  maxPoints: number
): { x: number[]; y: number[] } {
  if (xData.length <= maxPoints) {
    return { x: xData, y: yData };
  }

  const sampledX: number[] = [xData[0]];
  const sampledY: number[] = [yData[0]];

  // 始终保留第一个和最后一个点
  const bucketSize = (xData.length - 2) / (maxPoints - 2);

  let a = 0;

  for (let i = 0; i < maxPoints - 2; i++) {
    const avgRangeStart = Math.floor((i + 0) * bucketSize) + 1;
    const avgRangeEnd = Math.floor((i + 1) * bucketSize) + 1;
    const avgRangeLength = avgRangeEnd - avgRangeStart;

    // 计算下一个桶的平均点
    let maxArea = -1;
    let maxAreaPoint = avgRangeStart;

    const rangeStart = Math.floor((i + 0) * bucketSize) + 1;
    const rangeEnd = Math.floor((i + 1) * bucketSize) + 1;

    const pointAX = sampledX[a];
    const pointAY = sampledY[a];

    for (let j = rangeStart; j < rangeEnd; j++) {
      // 计算三角形面积
      const area = Math.abs(
        (pointAX - avgRangeStart) * (yData[j] - pointAY) -
        (pointAX - xData[j]) * (avgRangeStart - pointAY)
      ) * 0.5;

      if (area > maxArea) {
        maxArea = area;
        maxAreaPoint = j;
      }
    }

    sampledX.push(xData[maxAreaPoint]);
    sampledY.push(yData[maxAreaPoint]);
    a++;
  }

  sampledX.push(xData[xData.length - 1]);
  sampledY.push(yData[yData.length - 1]);

  return { x: sampledX, y: sampledY };
}

/**
 * 自适应采样
 * 根据数据特征选择最佳采样策略
 */
export function adaptiveSampling(
  xData: number[],
  yData: number[],
  maxPoints: number
): { x: number[]; y: number[] } {
  // 检测数据特征
  const characteristics = detectDataCharacteristics(yData);

  switch (characteristics) {
    case 'stable':
      // 平稳数据：等间隔采样
      return uniformSampling(xData, yData, maxPoints);

    case 'periodic':
      // 周期数据：峰值保留采样
      return peakPreservingSampling(xData, yData, maxPoints);

    case 'trend':
    case 'volatile':
      // 趋势/波动数据：LTTB采样
      return lttbSampling(xData, yData, maxPoints);

    default:
      return lttbSampling(xData, yData, maxPoints);
  }
}

/**
 * 检测数据特征
 */
function detectDataCharacteristics(
  data: number[]
): 'stable' | 'periodic' | 'trend' | 'volatile' {
  if (data.length < 10) return 'stable';

  // 计算统计量
  const mean = data.reduce((a, b) => a + b, 0) / data.length;
  const variance = data.reduce((a, b) => a + (b - mean) ** 2, 0) / data.length;
  const std = Math.sqrt(variance);
  const cv = std / Math.abs(mean); // 变异系数

  // 检测趋势
  const firstHalf = data.slice(0, data.length / 2);
  const secondHalf = data.slice(data.length / 2);
  const firstMean = firstHalf.reduce((a, b) => a + b, 0) / firstHalf.length;
  const secondMean = secondHalf.reduce((a, b) => a + b, 0) / secondHalf.length;
  const trendRatio = Math.abs(secondMean - firstMean) / std;

  // 判断特征
  if (cv < 0.1) {
    return 'stable';
  } else if (trendRatio > 1) {
    return 'trend';
  } else if (cv > 0.5) {
    return 'volatile';
  } else {
    return 'periodic';
  }
}

/**
 * 等间隔采样
 */
function uniformSampling(
  xData: number[],
  yData: number[],
  maxPoints: number
): { x: number[]; y: number[] } {
  const ratio = xData.length / maxPoints;
  const sampledX: number[] = [];
  const sampledY: number[] = [];

  for (let i = 0; i < maxPoints; i++) {
    const index = Math.floor(i * ratio);
    sampledX.push(xData[index]);
    sampledY.push(yData[index]);
  }

  return { x: sampledX, y: sampledY };
}

/**
 * 峰值保留采样
 */
function peakPreservingSampling(
  xData: number[],
  yData: number[],
  maxPoints: number
): { x: number[]; y: number[] } {
  // 简化实现：保留极值点
  const peaks = findPeaks(yData);
  const selectedIndices = new Set<number>();

  // 添加峰值
  peaks.forEach((index) => {
    if (selectedIndices.size < maxPoints) {
      selectedIndices.add(index);
    }
  });

  // 填充剩余点
  const remaining = maxPoints - selectedIndices.size;
  const step = Math.floor(xData.length / remaining);

  for (let i = 0; i < remaining && selectedIndices.size < maxPoints; i++) {
    selectedIndices.add(i * step);
  }

  // 排序并返回
  const sortedIndices = Array.from(selectedIndices).sort((a, b) => a - b);

  return {
    x: sortedIndices.map((i) => xData[i]),
    y: sortedIndices.map((i) => yData[i]),
  };
}

/**
 * 查找峰值
 */
function findPeaks(data: number[]): number[] {
  const peaks: number[] = [];

  for (let i = 1; i < data.length - 1; i++) {
    if (data[i] > data[i - 1] && data[i] > data[i + 1]) {
      peaks.push(i);
    } else if (data[i] < data[i - 1] && data[i] < data[i + 1]) {
      peaks.push(i);
    }
  }

  return peaks;
}
```

```typescript
/**
 * 请求去重器
 * 避免重复发送相同请求
 */
export class RequestDeduplicator {
  private pendingRequests: Map<string, Promise<any>> = new Map();

  /**
   * 执行去重请求
   */
  async request<T>(key: string, fetcher: () => Promise<T>): Promise<T> {
    // 检查是否有相同请求正在进行
    if (this.pendingRequests.has(key)) {
      return this.pendingRequests.get(key) as Promise<T>;
    }

    // 创建新请求
    const promise = fetcher().finally(() => {
      this.pendingRequests.delete(key);
    });

    this.pendingRequests.set(key, promise);

    return promise;
  }

  /**
   * 清除所有待处理请求
   */
  clear(): void {
    this.pendingRequests.clear();
  }
}

/**
 * 请求批处理器
 * 合并多个请求为批量请求
 */
export class RequestBatcher<T, R> {
  private queue: Array<{
    item: T;
    resolve: (value: R) => void;
    reject: (error: Error) => void;
  }> = [];

  private timer: number | null = null;

  constructor(
    private batchHandler: (items: T[]) => Promise<R[]>,
    private options: {
      maxBatchSize?: number;
      maxWaitTime?: number;
    } = {}
  ) {
    this.options = {
      maxBatchSize: 10,
      maxWaitTime: 100,
      ...options,
    };
  }

  /**
   * 添加请求到批处理队列
   */
  async add(item: T): Promise<R> {
    return new Promise((resolve, reject) => {
      this.queue.push({ item, resolve, reject });

      // 达到最大批次大小，立即执行
      if (this.queue.length >= (this.options.maxBatchSize || 10)) {
        this.flush();
      } else {
        // 设置定时器
        this.scheduleFlush();
      }
    });
  }

  /**
   * 调度刷新
   */
  private scheduleFlush(): void {
    if (this.timer) return;

    this.timer = window.setTimeout(() => {
      this.flush();
    }, this.options.maxWaitTime);
  }

  /**
   * 执行批处理
   */
  private async flush(): Promise<void> {
    if (this.timer) {
      clearTimeout(this.timer);
      this.timer = null;
    }

    const batch = this.queue.splice(0, this.queue.length);
    if (batch.length === 0) return;

    try {
      const items = batch.map((b) => b.item);
      const results = await this.batchHandler(items);

      batch.forEach((b, i) => {
        b.resolve(results[i]);
      });
    } catch (error) {
      batch.forEach((b) => {
        b.reject(error as Error);
      });
    }
  }
}

/**
 * 数据预加载器
 */
export class DataPreloader {
  private cache: Map<string, { data: any; timestamp: number }> = new Map();
  private preloadQueue: Array<() => Promise<void>> = [];

  /**
   * 预加载数据
   */
  preload(key: string, fetcher: () => Promise<any>, ttl: number = 60000): void {
    this.preloadQueue.push(async () => {
      try {
        const data = await fetcher();
        this.cache.set(key, {
          data,
          timestamp: Date.now() + ttl,
        });
      } catch (error) {
        console.error(`预加载失败 ${key}:`, error);
      }
    });

    this.processQueue();
  }

  /**
   * 获取预加载数据
   */
  get<T>(key: string): T | null {
    const cached = this.cache.get(key);

    if (cached && cached.timestamp > Date.now()) {
      return cached.data as T;
    }

    return null;
  }

  /**
   * 处理预加载队列
   */
  private async processQueue(): Promise<void> {
    while (this.preloadQueue.length > 0) {
      const task = this.preloadQueue.shift();
      if (task) {
        await task();
      }
    }
  }
}
```

```typescript
interface WebSocketMessage {

  data: any;

/**
 * WebSocket连接池
 * 管理多个WebSocket连接
 */
export class WebSocketPool {
  private connections: Map<string, WebSocket> = new Map();
  private messageQueue: Map<string, WebSocketMessage[]> = new Map();
  private maxQueueSize: number = 100;

  /**
   * 获取或创建连接
   */
  connect(url: string, options?: {
    onMessage?: (data: any) => void;
    onOpen?: () => void;
    onClose?: () => void;
    onError?: (error: Event) => void;
  }): WebSocket {
    if (this.connections.has(url)) {
      return this.connections.get(url)!;
    }

    const ws = new WebSocket(url);

    ws.onopen = () => {
      console.log(`WebSocket连接建立: ${url}`);
      options?.onOpen?.();

      // 发送队列中的消息
      this.flushQueue(url);
    };

    ws.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data);
        options?.onMessage?.(data);
      } catch (error) {
        console.error('WebSocket消息解析失败:', error);
      }
    };

    ws.onclose = () => {
      console.log(`WebSocket连接关闭: ${url}`);
      this.connections.delete(url);
      options?.onClose?.();
    };

    ws.onerror = (error) => {
      console.error(`WebSocket错误: ${url}`, error);
      options?.onError?.(error);
    };

    this.connections.set(url, ws);
    this.messageQueue.set(url, []);

    return ws;
  }

  /**
   * 发送消息（带队列）
   */
  send(url: string, message: any): void {
    const ws = this.connections.get(url);

    if (ws && ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify(message));
    } else {
      // 连接未就绪，加入队列
      const queue = this.messageQueue.get(url) || [];

      if (queue.length < this.maxQueueSize) {
        queue.push({
          type: 'send',
          data: message,
          timestamp: Date.now(),
        });
        this.messageQueue.set(url, queue);
      }
    }
  }

  /**
   * 刷新消息队列
   */
  private flushQueue(url: string): void {
    const ws = this.connections.get(url);
    const queue = this.messageQueue.get(url) || [];

    if (ws && ws.readyState === WebSocket.OPEN) {
      queue.forEach((msg) => {
        ws.send(JSON.stringify(msg.data));
      });

      this.messageQueue.set(url, []);
    }
  }

  /**
   * 关闭连接
   */
  disconnect(url: string): void {
    const ws = this.connections.get(url);

    if (ws) {
      ws.close();
      this.connections.delete(url);
      this.messageQueue.delete(url);
    }
  }

  /**
   * 关闭所有连接
   */
  disconnectAll(): void {
    this.connections.forEach((ws, url) => {
      ws.close();
    });

    this.connections.clear();
    this.messageQueue.clear();
  }
}

/**
 * 消息节流器
 * 控制消息推送频率
 */
export class MessageThrottler {
  private lastMessageTime: Map<string, number> = new Map();
  private pendingMessages: Map<string, any> = new Map();
  private timers: Map<string, number> = new Map();

  constructor(private minInterval: number = 100) {}

  /**
   * 节流处理消息
   */
  throttle(key: string, message: any, handler: (msg: any) => void): void {
    const now = Date.now();
    const lastTime = this.lastMessageTime.get(key) || 0;

    // 存储最新消息
    this.pendingMessages.set(key, message);

    // 检查是否可以立即处理
    if (now - lastTime >= this.minInterval) {
      this.processMessage(key, handler);
    } else {
      // 设置定时器
      this.scheduleProcess(key, handler);
    }
  }

  /**
   * 处理消息
   */
  private processMessage(key: string, handler: (msg: any) => void): void {
    const message = this.pendingMessages.get(key);

    if (message) {
      handler(message);
      this.lastMessageTime.set(key, Date.now());
      this.pendingMessages.delete(key);
    }
  }

  /**
   * 调度处理
   */
  private scheduleProcess(key: string, handler: (msg: any) => void): void {
    if (this.timers.has(key)) return;

    const timer = window.setTimeout(() => {
      this.processMessage(key, handler);
      this.timers.delete(key);
    }, this.minInterval);

    this.timers.set(key, timer);
  }

  /**
   * 清除所有定时器
   */
  clear(): void {
    this.timers.forEach((timer) => clearTimeout(timer));
    this.timers.clear();
    this.pendingMessages.clear();
  }
}

export const messageThrottler = new MessageThrottler();
```

```python
from typing import Optional
from dataclasses import dataclass
from sqlalchemy import create_engine, event
from sqlalchemy.orm import sessionmaker, Session
from sqlalchemy.pool import QueuePool
import logging

logger = logging.getLogger(__name__)


@dataclass
class PoolConfig:
    """连接池配置。"""
    pool_size: int = 5           # 连接池大小
    max_overflow: int = 10       # 最大溢出连接数
    pool_timeout: float = 30.0   # 获取连接超时(秒)
    pool_recycle: int = 3600     # 连接回收时间(秒)
    pool_pre_ping: bool = True   # 连接健康检查
    echo: bool = False           # SQL日志


class DatabasePool:
    """
    数据库连接池管理器。

    提供统一的数据库连接管理，支持SQLite和PostgreSQL。
    """

    def __init__(
        self,
        database_url: str,
        config: Optional[PoolConfig] = None,
    ):
        """
        初始化连接池。

        Args:
            database_url: 数据库URL
            config: 连接池配置
        """
        self.database_url = database_url
        self.config = config or PoolConfig()

        # 创建引擎
        self.engine = self._create_engine()

        # 创建会话工厂
        self.SessionLocal = sessionmaker(
            bind=self.engine,
            autocommit=False,
            autoflush=False,
        )

        # 配置SQLite优化
        if database_url.startswith('sqlite'):
            self._configure_sqlite()

    def _create_engine(self):
        """
        创建数据库引擎。

        Returns:
            SQLAlchemy引擎
        """
        # SQLite配置
        if self.database_url.startswith('sqlite'):
            return create_engine(
                self.database_url,
                connect_args={'check_same_thread': False},
                poolclass=QueuePool,
                pool_size=self.config.pool_size,
                max_overflow=self.config.max_overflow,
                pool_timeout=self.config.pool_timeout,
                pool_recycle=self.config.pool_recycle,
                pool_pre_ping=self.config.pool_pre_ping,
                echo=self.config.echo,
            )

        # PostgreSQL配置
        return create_engine(
            self.database_url,
            pool_size=self.config.pool_size,
            max_overflow=self.config.max_overflow,
            pool_timeout=self.config.pool_timeout,
            pool_recycle=self.config.pool_recycle,
            pool_pre_ping=self.config.pool_pre_ping,
            echo=self.config.echo,
        )

    def _configure_sqlite(self):
        """
        配置SQLite优化参数。
        """
        @event.listens_for(self.engine, "connect")
        def set_sqlite_pragma(dbapi_conn, connection_record):
            """设置SQLite PRAGMA参数。"""
            cursor = dbapi_conn.cursor()

            # 启用外键约束
            cursor.execute("PRAGMA foreign_keys=ON")

            # WAL模式（提高并发性能）
            cursor.execute("PRAGMA journal_mode=WAL")

            # 同步模式（平衡性能和安全）
            cursor.execute("PRAGMA synchronous=NORMAL")

            # 缓存大小（64MB）
            cursor.execute("PRAGMA cache_size=-64000")

            # 临时存储使用内存
            cursor.execute("PRAGMA temp_store=MEMORY")

            # 忙等待超时（30秒）
            cursor.execute("PRAGMA busy_timeout=30000")

            cursor.close()

            logger.debug("SQLite PRAGMA配置完成")

    def get_session(self) -> Session:
        """
        获取数据库会话。

        Returns:
            数据库会话
        """
        return self.SessionLocal()

    def get_statistics(self) -> dict:
        """
        获取连接池统计信息。

        Returns:
            统计信息字典
        """
        pool = self.engine.pool

        return {
            "pool_size": pool.size(),
            "checked_in": pool.checkedin(),
            "checked_out": pool.checkedout(),
            "overflow": pool.overflow(),
            "invalid": pool.invalidatedcount(),
        }

    def dispose(self):
        """释放所有连接。"""
        self.engine.dispose()


# 全局连接池实例
_db_pool: Optional[DatabasePool] = None


def init_database_pool(
    database_url: str,
    config: Optional[PoolConfig] = None,
) -> DatabasePool:
    """
    初始化全局数据库连接池。

    Args:
        database_url: 数据库URL
        config: 连接池配置

    Returns:
        连接池实例
    """
    global _db_pool
    _db_pool = DatabasePool(database_url, config)
    return _db_pool


def get_db_pool() -> Optional[DatabasePool]:
    """获取全局连接池实例。"""
```

```python
"""

功能: 查询优化工具

from typing import List, Optional, Dict, Any
from sqlalchemy.orm import Session, Query
from sqlalchemy import func, and_, or_
import logging

logger = logging.getLogger(__name__)


class QueryOptimizer:
    """
    查询优化器。

    提供查询优化和性能监控功能。
    """

    @staticmethod
    def optimize_pagination(
        query: Query,
        page: int,
        page_size: int,
    ) -> Query:
        """
        优化分页查询。

        使用延迟加载和游标分页提高性能。

        Args:
            query: SQLAlchemy查询对象
            page: 页码
            page_size: 每页数量

        Returns:
            优化后的查询
        """
        offset = (page - 1) * page_size
        return query.offset(offset).limit(page_size)

    @staticmethod
    def eager_load_relationships(
        query: Query,
        relationships: List[str],
    ) -> Query:
        """
        预加载关联关系。

        避免N+1查询问题。

        Args:
            query: SQLAlchemy查询对象
            relationships: 关联关系名称列表

        Returns:
            优化后的查询
        """
        from sqlalchemy.orm import joinedload

        for rel in relationships:
            query = query.options(joinedload(rel))

        return query

    @staticmethod
    def batch_insert(
        session: Session,
        model_class,
        data_list: List[Dict[str, Any]],
        batch_size: int = 100,
    ) -> int:
        """
        批量插入数据。

        使用批量插入提高性能。

        Args:
            session: 数据库会话
            model_class: 模型类
            data_list: 数据列表
            batch_size: 批次大小

        Returns:
            插入记录数
        """
        total_inserted = 0

        for i in range(0, len(data_list), batch_size):
            batch = data_list[i:i + batch_size]

            # 批量创建对象
            objects = [model_class(**data) for data in batch]

            # 批量添加
            session.bulk_save_objects(objects)
            session.commit()

            total_inserted += len(batch)

            logger.debug(f"批量插入: {len(batch)} 条记录")

        return total_inserted

    @staticmethod
    def optimize_count_query(query: Query) -> int:
        """
        优化计数查询。

        使用更高效的计数方式。

        Args:
            query: SQLAlchemy查询对象

        Returns:
            记录数
        """
        # 使用子查询计数
        count_query = query.with_entities(func.count()).statement

        return query.session.execute(count_query).scalar()
```

```python
"""
文件名: core/cache.py
功能: 缓存管理器
"""

import json
import hashlib
from typing import Optional, Any, Dict, Callable
from datetime import timedelta
from dataclasses import dataclass
import asyncio
import logging

logger = logging.getLogger(__name__)


@dataclass
class CacheConfig:
    """缓存配置。"""
    default_ttl: int = 300          # 默认TTL(秒)
    max_memory_items: int = 1000    # 内存缓存最大条目数
    key_prefix: str = "cauc_sep:"   # 键前缀
    enable_redis: bool = False      # 是否启用Redis
    redis_url: str = "redis://localhost:6379/0"


class MemoryCache:
    """
    内存缓存实现。

    简单高效的进程内缓存。
    """

    def __init__(self, max_items: int = 1000):
        """
        初始化内存缓存。

        Args:
            max_items: 最大缓存条目数
        """
        self._cache: Dict[str, tuple] = {}  # {key: (value, expire_time)}
        self._max_items = max_items

    async def get(self, key: str) -> Optional[Any]:
        """
        获取缓存值。

        Args:
            key: 缓存键

        Returns:
            缓存值，不存在或过期返回None
        """
        import time

        if key not in self._cache:
            return None

        value, expire_time = self._cache[key]

        # 检查是否过期
        if expire_time and time.time() > expire_time:
            del self._cache[key]
            return None

        return value

    async def set(
        self,
        key: str,
        value: Any,
        ttl: Optional[int] = None,
    ) -> None:
        """
        设置缓存值。

        Args:
            key: 缓存键
            value: 缓存值
            ttl: 过期时间(秒)
        """
        import time

        # 检查容量
        if len(self._cache) >= self._max_items:
            # 删除最旧的条目
            oldest_key = next(iter(self._cache))
            del self._cache[oldest_key]

        expire_time = time.time() + ttl if ttl else None
        self._cache[key] = (value, expire_time)

    async def delete(self, key: str) -> None:
        """
        删除缓存值。

        Args:
            key: 缓存键
        """
        self._cache.pop(key, None)

    async def clear(self) -> None:
        """清空缓存。"""
        self._cache.clear()


class RedisCache:
    """
    Redis缓存实现。

    分布式缓存，支持跨进程共享。
    """

    def __init__(self, redis_url: str):
        """
        初始化Redis缓存。

        Args:
            redis_url: Redis连接URL
        """
        self._redis_url = redis_url
        self._redis = None

    async def _get_redis(self):
        """获取Redis连接。"""
        if self._redis is None:
            try:
                import aioredis
                self._redis = await aioredis.from_url(
                    self._redis_url,
                    encoding="utf-8",
                    decode_responses=True,
                )
            except ImportError:
                logger.warning("aioredis未安装，Redis缓存不可用")
                raise
        return self._redis

    async def get(self, key: str) -> Optional[Any]:
        """
        获取缓存值。

        Args:
            key: 缓存键

        Returns:
            缓存值
        """
        try:
            redis = await self._get_redis()
            value = await redis.get(key)

            if value:
                return json.loads(value)
            return None

        except Exception as e:
            logger.error(f"Redis获取失败: {e}")
            return None

    async def set(
        self,
        key: str,
        value: Any,
        ttl: Optional[int] = None,
    ) -> None:
        """
        设置缓存值。

        Args:
            key: 缓存键
            value: 缓存值
            ttl: 过期时间(秒)
        """
        try:
            redis = await self._get_redis()
            serialized = json.dumps(value)

            if ttl:
                await redis.setex(key, ttl, serialized)
            else:
                await redis.set(key, serialized)

        except Exception as e:
            logger.error(f"Redis设置失败: {e}")

    async def delete(self, key: str) -> None:
        """
        删除缓存值。

        Args:
            key: 缓存键
        """
        try:
            redis = await self._get_redis()
            await redis.delete(key)
        except Exception as e:
            logger.error(f"Redis删除失败: {e}")

    async def clear(self) -> None:
        """清空缓存。"""
        try:
            redis = await self._get_redis()
            await redis.flushdb()
        except Exception as e:
            logger.error(f"Redis清空失败: {e}")


class CacheManager:
    """
    缓存管理器。

    统一管理多级缓存。
    """

    def __init__(self, config: Optional[CacheConfig] = None):
        """
        初始化缓存管理器。

        Args:
            config: 缓存配置
        """
        self.config = config or CacheConfig()

        # L1: 内存缓存
        self._memory_cache = MemoryCache(self.config.max_memory_items)

        # L2: Redis缓存（可选）
        self._redis_cache: Optional[RedisCache] = None

    async def initialize(self) -> None:
        """初始化缓存。"""
        if self.config.enable_redis:
            try:
                self._redis_cache = RedisCache(self.config.redis_url)
                logger.info("Redis缓存初始化成功")
            except Exception as e:
                logger.warning(f"Redis缓存初始化失败: {e}")

    def _make_key(self, key: str) -> str:
        """
        生成缓存键。

        Args:
            key: 原始键

        Returns:
            完整缓存键
        """
        return f"{self.config.key_prefix}{key}"

    async def get(
        self,
        key: str,
        default: Any = None,
    ) -> Any:
        """
        获取缓存值。

        依次从L1、L2缓存查找。

        Args:
            key: 缓存键
            default: 默认值

        Returns:
            缓存值
        """
        full_key = self._make_key(key)

        # L1: 内存缓存
        value = await self._memory_cache.get(full_key)
        if value is not None:
            return value

        # L2: Redis缓存
        if self._redis_cache:
            value = await self._redis_cache.get(full_key)
            if value is not None:
                # 回填L1缓存
                await self._memory_cache.set(full_key, value, self.config.default_ttl)
                return value

        return default

    async def set(
        self,
        key: str,
        value: Any,
        ttl: Optional[int] = None,
    ) -> None:
        """
        设置缓存值。

        同时写入L1和L2缓存。

        Args:
            key: 缓存键
            value: 缓存值
            ttl: 过期时间(秒)
        """
        full_key = self._make_key(key)
        ttl = ttl or self.config.default_ttl

        # L1: 内存缓存
        await self._memory_cache.set(full_key, value, ttl)

        # L2: Redis缓存
        if self._redis_cache:
            await self._redis_cache.set(full_key, value, ttl)

    async def delete(self, key: str) -> None:
        """
        删除缓存值。

        Args:
            key: 缓存键
        """
        full_key = self._make_key(key)

        await self._memory_cache.delete(full_key)

        if self._redis_cache:
            await self._redis_cache.delete(full_key)

    async def get_or_set(
        self,
        key: str,
        factory: Callable[[], Any],
        ttl: Optional[int] = None,
    ) -> Any:
        """
        获取或设置缓存值。

        如果缓存不存在，调用factory生成值并缓存。

        Args:
            key: 缓存键
            factory: 值生成函数
            ttl: 过期时间(秒)

        Returns:
            缓存值
        """
        value = await self.get(key)

        if value is None:
            value = await factory() if asyncio.iscoroutinefunction(factory) else factory()
            await self.set(key, value, ttl)

        return value

    async def clear(self) -> None:
        """清空所有缓存。"""
        await self._memory_cache.clear()

        if self._redis_cache:
            await self._redis_cache.clear()


# 全局缓存管理器
_cache_manager: Optional[CacheManager] = None


def init_cache_manager(config: Optional[CacheConfig] = None) -> CacheManager:
    """
    初始化全局缓存管理器。

    Args:
        config: 缓存配置

    Returns:
        缓存管理器实例
    """
    global _cache_manager
    _cache_manager = CacheManager(config)
    return _cache_manager


def get_cache_manager() -> Optional[CacheManager]:
    """获取全局缓存管理器。"""
    return _cache_manager
```

```python
"""
文件名: core/cache_decorator.py
功能: 缓存装饰器

from functools import wraps
from typing import Optional, Callable, Any
import hashlib
import json
import logging

from core.cache import get_cache_manager

logger = logging.getLogger(__name__)


def cache_result(
    key_template: Optional[str] = None,
    ttl: int = 300,
    key_params: Optional[list] = None,
):
    """
    缓存函数结果装饰器。

    Args:
        key_template: 缓存键模板，支持{param}占位符
        ttl: 缓存过期时间(秒)
        key_params: 用于生成键的参数名列表

    Example:
        @cache_result(key_template="user:{user_id}", ttl=60)
        async def get_user(user_id: int):
            ...
    """
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        async def async_wrapper(*args, **kwargs) -> Any:
            cache = get_cache_manager()
            if not cache:
                return await func(*args, **kwargs)

            # 生成缓存键
            cache_key = _generate_cache_key(
                func, key_template, key_params, args, kwargs
            )

            # 尝试从缓存获取
            cached_value = await cache.get(cache_key)
            if cached_value is not None:
                logger.debug(f"缓存命中: {cache_key}")
                return cached_value

            # 执行函数
            result = await func(*args, **kwargs)

            # 缓存结果
            await cache.set(cache_key, result, ttl)
            logger.debug(f"缓存设置: {cache_key}")

            return result

        @wraps(func)
        def sync_wrapper(*args, **kwargs) -> Any:
            import asyncio

            cache = get_cache_manager()
            if not cache:
                return func(*args, **kwargs)

            # 生成缓存键
            cache_key = _generate_cache_key(
                func, key_template, key_params, args, kwargs
            )

            # 尝试从缓存获取
            loop = asyncio.get_event_loop()
            cached_value = loop.run_until_complete(cache.get(cache_key))

            if cached_value is not None:
                logger.debug(f"缓存命中: {cache_key}")
                return cached_value

            # 执行函数
            result = func(*args, **kwargs)

            # 缓存结果
            loop.run_until_complete(cache.set(cache_key, result, ttl))
            logger.debug(f"缓存设置: {cache_key}")

            return result

        import asyncio
        if asyncio.iscoroutinefunction(func):
            return async_wrapper
        return sync_wrapper

    return decorator


def _generate_cache_key(
    func: Callable,
    key_template: Optional[str],
    key_params: Optional[list],
    args: tuple,
    kwargs: dict,
) -> str:
    """
    生成缓存键。

    Args:
        func: 函数对象
        key_template: 键模板
        key_params: 键参数
        args: 位置参数
        kwargs: 关键字参数

    Returns:
        缓存键
    """
    if key_template:
        # 使用模板生成键
        # 获取函数参数
        import inspect
        sig = inspect.signature(func)
        bound = sig.bind(*args, **kwargs)
        bound.apply_defaults()

        # 替换占位符
        key = key_template.format(**bound.arguments)
        return key

    elif key_params:
        # 使用指定参数生成键
        import inspect
        sig = inspect.signature(func)
        bound = sig.bind(*args, **kwargs)
        bound.apply_defaults()

        key_parts = [func.__name__]
        for param in key_params:
            if param in bound.arguments:
                key_parts.append(str(bound.arguments[param]))

        return ":".join(key_parts)

    else:
        # 使用所有参数生成键
        key_data = {
            "func": func.__name__,
            "args": args,
            "kwargs": kwargs,
        }
        key_str = json.dumps(key_data, sort_keys=True)
        key_hash = hashlib.md5(key_str.encode()).hexdigest()
        return f"{func.__name__}:{key_hash}"
```

```python
"""

功能: 性能分析器
"""

import time
import cProfile
import pstats
import io
from typing import Optional, Dict, Any, List
from dataclasses import dataclass, field
from functools import wraps
import logging

logger = logging.getLogger(__name__)


@dataclass
class PerformanceMetric:
    """性能指标数据类。"""
    name: str
    duration_ms: float
    memory_mb: float = 0.0
    cpu_percent: float = 0.0
    timestamp: float = field(default_factory=time.time)
    metadata: Dict[str, Any] = field(default_factory=dict)


class PerformanceProfiler:
    """
    性能分析器。

    提供函数级别性能追踪和分析功能。
    """

    def __init__(self):
        """初始化分析器。"""
        self._metrics: List[PerformanceMetric] = []
        self._profiler: Optional[cProfile.Profile] = None

    def start_profiling(self) -> None:
        """开始性能分析。"""
        self._profiler = cProfile.Profile()
        self._profiler.enable()

    def stop_profiling(self) -> Dict[str, Any]:
        """
        停止性能分析。

        Returns:
            分析结果
        """
        if not self._profiler:
            return {}

        self._profiler.disable()

        # 生成统计信息
        s = io.StringIO()
        ps = pstats.Stats(self._profiler, stream=s).sort_stats('cumulative')
        ps.print_stats(20)

        result = {
            "stats": s.getvalue(),
            "total_calls": ps.total_calls,
            "total_time": ps.total_tt,
        }

        self._profiler = None
        return result

    def record_metric(self, metric: PerformanceMetric) -> None:
        """
        记录性能指标。

        Args:
            metric: 性能指标
        """
        self._metrics.append(metric)

    def get_statistics(self) -> Dict[str, Any]:
        """
        获取性能统计信息。

        Returns:
            统计信息
        """
        if not self._metrics:
            return {}

        durations = [m.duration_ms for m in self._metrics]

        return {
            "total_metrics": len(self._metrics),
            "avg_duration_ms": sum(durations) / len(durations),
            "min_duration_ms": min(durations),
            "max_duration_ms": max(durations),
        }


def profile_function(name: str):
    """
    函数性能分析装饰器。

    Args:
        name: 分析名称

    Example:
        @profile_function("data_processing")
        async def process_data(data):
            ...
    """
    def decorator(func):
        @wraps(func)
        async def async_wrapper(*args, **kwargs):
            import psutil
            import os

            process = psutil.Process(os.getpid())

            # 记录开始状态
            start_time = time.perf_counter()
            start_memory = process.memory_info().rss / 1024 / 1024

            try:
                result = await func(*args, **kwargs)
                return result
            finally:
                # 记录结束状态
                end_time = time.perf_counter()
                end_memory = process.memory_info().rss / 1024 / 1024

                metric = PerformanceMetric(
                    name=name,
                    duration_ms=(end_time - start_time) * 1000,
                    memory_mb=end_memory - start_memory,
                )

                profiler = get_profiler()
                profiler.record_metric(metric)

                logger.debug(
                    f"性能分析 [{name}]: "
                    f"耗时={metric.duration_ms:.2f}ms, "
                    f"内存={metric.memory_mb:.2f}MB"
                )

        @wraps(func)
        def sync_wrapper(*args, **kwargs):
            import psutil
            import os

            process = psutil.Process(os.getpid())

            start_time = time.perf_counter()
            start_memory = process.memory_info().rss / 1024 / 1024

            try:
                result = func(*args, **kwargs)
                return result
            finally:
                end_time = time.perf_counter()
                end_memory = process.memory_info().rss / 1024 / 1024

                metric = PerformanceMetric(
                    name=name,
                    duration_ms=(end_time - start_time) * 1000,
                    memory_mb=end_memory - start_memory,
                )

                profiler = get_profiler()
                profiler.record_metric(metric)

        import asyncio
        if asyncio.iscoroutinefunction(func):
            return async_wrapper
        return sync_wrapper

    return decorator


# 全局分析器实例
_profiler: Optional[PerformanceProfiler] = None


def get_profiler() -> PerformanceProfiler:
    """获取全局分析器实例。"""
    global _profiler
    if _profiler is None:
        _profiler = PerformanceProfiler()
    return _profiler
```

```python
"""
文件名: api/performance.py
功能: 性能监控API

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import List, Dict, Any

from core.profiler import get_profiler

router = APIRouter(prefix="/api/v1/performance", tags=["performance"])


class PerformanceMetricResponse(BaseModel):
    """性能指标响应模型。"""
    name: str
    duration_ms: float
    memory_mb: float
    timestamp: float


@router.get("/metrics", summary="获取性能指标")
async def get_performance_metrics() -> List[PerformanceMetricResponse]:
    """
    获取所有性能指标。

    Returns:
        性能指标列表
    """
    profiler = get_profiler()
    return [
        PerformanceMetricResponse(
            name=m.name,
            duration_ms=m.duration_ms,
            memory_mb=m.memory_mb,
            timestamp=m.timestamp,
        )
        for m in profiler._metrics
    ]


@router.get("/statistics", summary="获取性能统计")
async def get_performance_statistics() -> Dict[str, Any]:
    """
    获取性能统计信息。

    Returns:
        统计信息
    """
    profiler = get_profiler()
    return profiler.get_statistics()


@router.post("/profile/start", summary="开始性能分析")
async def start_profiling():
    """开始性能分析。"""
    profiler = get_profiler()
    profiler.start_profiling()
    return {"message": "性能分析已开始"}


@router.post("/profile/stop", summary="停止性能分析")
async def stop_profiling() -> Dict[str, Any]:
    """
    停止性能分析。

    Returns:
        分析结果
    """
    profiler = get_profiler()
```
