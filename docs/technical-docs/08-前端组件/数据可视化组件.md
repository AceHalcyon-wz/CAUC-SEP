# 数据可视化组件

## 概述

数据可视化组件提供图表展示、数据处理和分析功能，基于 ECharts 实现，支持多种图表类型、交互式缩放平移、数据标注等功能。

### 功能特性

- 多种图表类型：折线图、散点图、柱状图、面积图、混合图

```vue
  }
```

### 缩放功能

```javascript
// 缩放放大
function handleZoomIn() {
  const option = chartInstance.getOption()
  const currentStart = option.dataZoom[0].start || 0
  const currentEnd = option.dataZoom[0].end || 100
  const range = currentEnd - currentStart
  
  const newRange = range * 0.8
  const center = (currentStart + currentEnd) / 2
  const newStart = Math.max(0, center - newRange / 2)
  const newEnd = Math.min(100, center + newRange / 2)
  
  animateZoom(currentStart, currentEnd, newStart, newEnd)
}

// 缩放缩小
function handleZoomOut() {
  const option = chartInstance.getOption()
  const currentStart = option.dataZoom[0].start || 0
  const currentEnd = option.dataZoom[0].end || 100
  const range = currentEnd - currentStart
  const newRange = Math.min(100, range * 1.25)
  const center = (currentStart + currentEnd) / 2
  const newStart = Math.max(0, center - newRange / 2)
  const newEnd = Math.min(100, center + newRange / 2)
  
  animateZoom(currentStart, currentEnd, newStart, newEnd)
}

// 平滑缩放动画
function animateZoom(fromStart, fromEnd, toStart, toEnd) {
  const duration = 300
  const startTime = Date.now()
  
  function animate() {
    const elapsed = Date.now() - startTime
    const progress = Math.min(elapsed / duration, 1)
    const eased = easeOutCubic(progress)
    
    const currentStart = fromStart + (toStart - fromStart) * eased
    const currentEnd = fromEnd + (toEnd - fromEnd) * eased
    
    chartInstance.dispatchAction({
      type: 'dataZoom',
      start: currentStart,
      end: currentEnd
    })
    
    if (progress < 1) {
      requestAnimationFrame(animate)
    }
  }
  
  requestAnimationFrame(animate)
}
```

```typescript
function addPointAnnotation(dataIndex, value, name) {
  const annotation = {

    type: 'point',

      coord: [dataIndex, value],
      value: value.toFixed(3)
    },

    color: '#f56c6c'
  }
  
  annotations.value.push(annotation)
  updateChartAnnotations()
}

// 添加线标注
function handleLineAnnotation(dataIndex, value) {
  if (!tempAnnotation.isDrawing) {
    tempAnnotation.startPoint = { dataIndex, value }
    tempAnnotation.isDrawing = true
    ElMessage.info('请点击第二个点完成线标注')
  } else {
    const annotation = {
      id: `line_${Date.now()}`,
      type: 'line',
      data: {
        start: tempAnnotation.startPoint,
        end: { dataIndex, value }
      },
      label: `线标注 ${annotations.value.length + 1}`,
      color: '#409eff'
    }
    
    annotations.value.push(annotation)
    updateChartAnnotations()
    tempAnnotation.startPoint = null
    tempAnnotation.isDrawing = false
  }
}

// 距离测量
function handleMeasureAnnotation(dataIndex, value) {
  if (!tempAnnotation.isDrawing) {
    tempAnnotation.startPoint = { dataIndex, value }
    tempAnnotation.isDrawing = true
    ElMessage.info('请点击第二个点测量距离')
  } else {
    const dx = Math.abs(dataIndex - tempAnnotation.startPoint.dataIndex)
    const dy = Math.abs(value - tempAnnotation.startPoint.value)
    const distance = Math.sqrt(dx * dx + dy * dy)
    
    const annotation = {
      id: `measure_${Date.now()}`,
      type: 'measure',
      data: {
        start: tempAnnotation.startPoint,
        end: { dataIndex, value },
        distance: distance.toFixed(3)
      },
      label: `距离: ${distance.toFixed(3)}`,
      color: '#e6a23c'
    }
    
    annotations.value.push(annotation)
    updateChartAnnotations()
    tempAnnotation.startPoint = null
    tempAnnotation.isDrawing = false
  }
}
```

```javascript
function handleImportConfig() {
  const input = document.createElement('input')
  input.type = 'file'
  input.accept = '.json'
  
  input.onchange = (e) => {
    const file = e.target.files[0]
    const reader = new FileReader()
    reader.onload = (event) => {
      const config = JSON.parse(event.target.result)
      chartType.value = config.chartType
      showGrid.value = config.showGrid
      enableZoom.value = config.enableZoom
      annotations.value = config.annotations || []
      updateChart()
      ElMessage.success('配置已导入')
    }
    reader.readAsText(file)
  }
  
  input.click()
}
```

## 性能优化

### 大数据量处理

```javascript
// 数据采样
function downsampleData(data, maxPoints) {
  if (data.length <= maxPoints) return data
  
  const step = Math.ceil(data.length / maxPoints)
  const sampled = []
  
  for (let i = 0; i < data.length; i += step) {
    sampled.push(data[i])
  }
  
  return sampled
}

// 智能采样（保留关键点）
function smartSampling(data, maxPoints) {
  if (data.length <= maxPoints) return data
  
  // 保留极值点
  const peaks = findPeaks(data)
  const valleys = findValleys(data)
  
  // 均匀采样其余点
  const remaining = maxPoints - peaks.length - valleys.length
  const uniform = uniformSample(data, remaining)
  
  return mergeAndSort([...peaks, ...valleys, ...uniform])
}
```

### 图表性能监控

```javascript
// FPS 监控
class FPSMonitor {
  constructor() {
    this.frames = []
    this.lastTime = performance.now()
  }
  
  tick() {
    const now = performance.now()
    const fps = 1000 / (now - this.lastTime)
    this.frames.push(fps)
    this.lastTime = now
    
    // 保留最近60帧
    if (this.frames.length > 60) {
      this.frames.shift()
    }
  }
  
  getAverageFPS() {
    return this.frames.reduce((a, b) => a + b, 0) / this.frames.length
  }
}

// 性能测量
function measureChartPerformance(chartInstance) {
  const start = performance.now()
  chartInstance.setOption(option)
  const end = performance.now()
  
  return {
    renderTime: end - start,
    dataPoints: option.series[0].data.length,
    fps: fpsMonitor.getAverageFPS()
  }
}
```
