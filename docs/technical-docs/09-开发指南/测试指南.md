## 1. 测试策略概述

### 1.1 测试金字塔

```
        /\
       /  \      E2E测试（少量）
      /----\     
     /      \    集成测试（适量）
    /--------\
   /          \  单元测试（大量）
  /------------\
```

### 1.2 测试类型

| 测试类型 | 工具 | 覆盖范围 | 运行频率 |
|----------|------|----------|----------|
| 单元测试 | pytest / Vitest | 函数、类方法 | 每次提交 |
| 集成测试 | pytest / Vitest | 模块间交互 | 每次合并 |
| E2E测试 | Playwright | 完整用户流程 | 每日/发布前 |

### 1.3 测试目标

| 指标 | 目标值 | 说明 |
|------|--------|------|
| 单元测试覆盖率 | >= 70% | 核心模块 |
| 集成测试覆盖率 | >= 50% | API端点 |
| E2E测试覆盖 | 关键流程 | 用户核心操作 |

```python
# tests/test_laser_driver.py
"""
激光器驱动单元测试
"""
import pytest
import numpy as np
from core.laser_driver import LaserDriver
from core.abstract import DeviceStatus


@pytest.fixture
def laser():
    """创建激光器实例（仿真模式）"""
    return LaserDriver("laser_01", {
        "simulation": True,
        "wavelength": 532,
        "max_power": 100,
    })


class TestLaserDriver:
    """激光器驱动测试类"""
    
    @pytest.mark.asyncio
    async def test_connect(self, laser):
        """测试连接"""
        result = await laser.connect()
        assert result is True
        assert laser.status == DeviceStatus.READY
    
    @pytest.mark.asyncio
    async def test_disconnect(self, laser):
        """测试断开连接"""
        await laser.connect()
        result = await laser.disconnect()
        assert result is True
        assert laser.status == DeviceStatus.DISCONNECTED
    
    @pytest.mark.asyncio
    async def test_set_power(self, laser):
        """测试设置功率"""
        await laser.connect()
        
        result = await laser.set_power(50.0)
        assert result is True
        
        status = await laser.read_status()
        assert status["current_power"] == 50.0
    
    @pytest.mark.asyncio
    async def test_set_power_invalid(self, laser):
        """测试无效功率"""
        await laser.connect()
        
        with pytest.raises(ValueError, match="Power must be between"):
            await laser.set_power(150.0)
    
    @pytest.mark.asyncio
    async def test_set_power_negative(self, laser):
        """测试负功率"""
        await laser.connect()
        
        with pytest.raises(ValueError):
            await laser.set_power(-10.0)
    
    @pytest.mark.asyncio
    async def test_read_status(self, laser):
        """测试读取状态"""
        await laser.connect()
        
        status = await laser.read_status()
        
        assert status["device_id"] == "laser_01"
        assert status["status"] == DeviceStatus.READY.value
        assert status["wavelength"] == 532
        assert status["connected"] is True
```

```typescript
import { describe, it, expect, beforeEach, vi } from 'vitest';
import { setActivePinia, createPinia } from 'pinia';
import { useDeviceStore } from '@/stores/device';

// Mock API
vi.mock('@/api/device', () => ({
  deviceApi: {
    getStatus: vi.fn(),
    connect: vi.fn(),
    disconnect: vi.fn(),
  },
}));

describe('DeviceStore', () => {
  beforeEach(() => {
    setActivePinia(createPinia());
  });

  it('initializes with empty devices', () => {
    const store = useDeviceStore();
    
    expect(store.devices).toEqual({});
    expect(store.isLoading).toBe(false);
  });

  it('fetches device status', async () => {
    const store = useDeviceStore();
    const mockStatus = { id: 'laser_01', status: 'ready' };
    
    vi.mocked(deviceApi.getStatus).mockResolvedValue(mockStatus);
    
    await store.fetchStatus('laser_01');
    
    expect(store.devices['laser_01']).toEqual(mockStatus);
  });

  it('handles connection error', async () => {
    const store = useDeviceStore();
    
    vi.mocked(deviceApi.connect).mockRejectedValue(new Error('Connection failed'));
    
    await expect(store.connect('laser_01')).rejects.toThrow('Connection failed');
    expect(store.error).toBe('Connection failed');
  });
```

## 6. 持续集成

### 6.1 GitHub Actions配置

```yaml
# .github/workflows/test.yml
name: Test

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main, develop]

jobs:
  backend-test:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      
      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'
          cache: 'pip'
      
      - name: Install dependencies
        run: |
          cd backend
          pip install -e ".[dev]"
      
      - name: Run linting
        run: |
          cd backend
          ruff check .
          mypy core api
      
      - name: Run tests
        run: |
          cd backend
          pytest --cov=core --cov=api --cov-report=xml
      
      - name: Upload coverage
        uses: codecov/codecov-action@v4
        with:
          files: backend/coverage.xml
          flags: backend

  frontend-test:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      
      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '18'
          cache: 'npm'
          cache-dependency-path: frontend/package-lock.json
      
      - name: Install dependencies
        run: |
          cd frontend
          npm ci
      
      - name: Run linting
        run: |
          cd frontend
          npm run lint:check
      
      - name: Run tests
        run: |
          cd frontend
          npm run test:coverage
      
      - name: Upload coverage
        uses: codecov/codecov-action@v4
        with:
          files: frontend/coverage/lcov.info
          flags: frontend

  e2e-test:
    runs-on: windows-latest
    needs: [backend-test, frontend-test]
    steps:
      - uses: actions/checkout@v4
      
      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '18'
      
      - name: Install Playwright
        run: |
          cd frontend
          npm ci
          npx playwright install --with-deps
      
      - name: Run E2E tests
        run: |
          cd frontend
          npm run test:e2e
      
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: playwright-report
          path: frontend/playwright-report/
          retention-days: 30
```
